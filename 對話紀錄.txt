[!CAUTION]
The prior mitigation only handled cases where the API raised an exception. It did not protect against the more severe failure mode in which the upstream call hangs forever without raising.
The required fix is a hard timeout wrapper using Python threading, matching the defensive pattern already used in stt_runner.py.

Proposed Changes
[MODIFY] merge_transcript.py
The implementation shall introduce a hard timeout wrapper inside merge_transcript.py so that both Vertex AI and Gemini HTTP fallback calls are strictly bounded by the configured timeout value.

This change ensures that a silent upstream hang is converted into a local TimeoutError, allowing the existing retry and degradation path to execute instead of permanently occupying worker capacity.

Required code changes
Add hard timeout helper: Implement _call_with_timeout() using threading.Thread, a result/error container, and Event.wait(timeout).

Wrap Vertex AI synchronous calls: Replace direct client.models.generate_content(...) execution with _call_with_timeout(...).

Wrap Gemini HTTP synchronous calls: Replace direct requests.post(...) execution with _call_with_timeout(...) so AI Studio or HTTP fallback paths cannot hang a worker indefinitely.

Add anti-reentry protection: Introduce an in-flight guard at the chunk or task level so the same unit of work cannot be sent to the LLM path by multiple scripts or threads concurrently. A process-wide IN_FLIGHT_TASKS set protected by a lock.

A manifest-level state transition from pending to running enforced atomically before any LLM call.

A per-chunk lock keyed by task_id + chunk_id.

def _call_with_timeout(func, timeout_sec, *args, **kwargs):
"""
執行帶有強制超時機制的函數呼叫，避免底層 C 函式庫或網路請求造成的無聲死鎖。
(參考自 stt_runner.py 的實作)
"""
import threading
result_box = [None]
error_box = [None]
done_evt = threading.Event()
def _target():
try:
result_box[0] = func(*args, **kwargs)
except Exception as e:
error_box[0] = e
finally:
done_evt.set()
t = threading.Thread(target=_target, daemon=True)
t.start()
fired = done_evt.wait(timeout_sec)

if not fired:
raise TimeoutError(f"{func.__name__} 逾時 {timeout_sec}s，放棄此呼叫並強制中斷")

if error_box[0]:
raise error_box[0]

return result_box[0]

def atomic_json_dump(data, filepath: str) -> None:
    """
    原子寫入 JSON，避免 Windows AV / 索引服務導致中間狀態被讀到。
    """
    tmp_path = f"{filepath}.{os.getpid()}.{threading.get_ident()}.tmp"
    with open(tmp_path, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)
        f.flush()
        try:
            os.fsync(f.fileno())
        except Exception:
            pass

    max_retries = 30
    for i in range(max_retries):
        try:
            os.replace(tmp_path, filepath)
            break
        except PermissionError:
            if i == max_retries - 1:
                logging.error(
                    f"嚴重 I/O 錯誤：等待 {max_retries} 次後仍無法覆寫 {filepath}"
                )
                if os.path.exists(tmp_path):
                    os.remove(tmp_path)
                raise
            import random

            sleep_time = (0.1 * (1.2 ** i)) + random.uniform(0.01, 0.05)
            time.sleep(min(sleep_time, 2.0))


def call_with_timeout(func, timeout_sec, *args, **kwargs):
    """
    在獨立 thread 中執行 func，若超時則拋出 TimeoutError。
    """
    result_box = {"result": None, "error": None}
    done_evt = threading.Event()

    def target():
        try:
            result_box["result"] = func(*args, **kwargs)
        except Exception as e:
            result_box["error"] = e
        finally:
            done_evt.set()

    t = threading.Thread(target=target, daemon=True)
    t.start()
    fired = done_evt.wait(timeout_sec)
    if not fired:
        raise TimeoutError(f"{func.__name__} timeout={timeout_sec}s")
    if result_box["error"] is not None:
        raise result_box["error"]
    return result_box["result"]


# ── Manifest 背景 writer：解決大量 chunk 更新的 I/O 壓力 ──


def manifest_writer_thread(chunks_manifest_path: str, manifest_update_queue: queue.Queue):
    """
    背景 manifest writer：批次把更新佇列裡的狀態刷新到 chunks manifest。
    這讓 process_single_chunk 可以把更新放到 queue，而不必每次都直接寫檔。
    """
    logging.info("ManifestWriter: Background writer started.")
    while True:
        try:
            item = manifest_update_queue.get(timeout=2.0)
        except queue.Empty:
            continue

        if item == "STOP":
            manifest_update_queue.task_done()
            break

        updates = [item]
        stop_seen = False

        # 把佇列裡現有更新一次取出
        while not manifest_update_queue.empty():
            try:
                val = manifest_update_queue.get_nowait()
                if val == "STOP":
                    stop_seen = True
                else:
                    updates.append(val)
            except queue.Empty:
                break

        if not updates:
            if stop_seen:
                manifest_update_queue.task_done()
                break
            continue

        # 讀出當前 chunks manifest
        try:
            if os.path.exists(chunks_manifest_path):
                with open(chunks_manifest_path, "r", encoding="utf-8-sig") as f:
                    current_chunks = json.load(f)
            else:
                current_chunks = []
        except Exception:
            current_chunks = []

        index = {}
        for c in current_chunks:
            if isinstance(c, dict):
                filename = c.get("filename")
                if filename:
                    index[filename] = c

        # 套用更新
        for u in updates:
            if not isinstance(u, dict):
                continue
            filename = u.get("filename")
            status = u.get("status")
            if not filename or not status:
                continue

            existing = index.get(filename)
            if not existing:
                existing = {"filename": filename}
                index[filename] = existing

            if "status" in u:
                existing["status"] = u["status"]
            if "started_at" in u:
                existing["started_at"] = u["started_at"]
            if "updated_at" in u:
                existing["updated_at"] = u["updated_at"]
            if "failed_at" in u:
                existing["failed_at"] = u["failed_at"]
            if "retry_count" in u:
                existing["retry_count"] = u["retry_count"]
            if "last_error" in u:
                existing["last_error"] = u["last_error"]

        # 寫回 updated chunks manifest
        atomic_json_dump(list(index.values()), chunks_manifest_path)

        for _ in updates:
            manifest_update_queue.task_done()

        if stop_seen:
            manifest_update_queue.task_done()
            break


# ── Distill worker：Whisper 與 Gemini 差異對比 ──


def distill_worker_thread(worker_id: int, distill_task_queue: queue.Queue):
    """
    背景 distill worker：用 WhisperPool 再跑一次本地轉錄，對照 Gemini 結果，寫 distill.json。
    """
    N = WhisperPool
    logging.info(f"DistillWorker-{worker_id}: Background distill worker started.")

    while True:
        task = distill_task_queue.get()
        if task == "STOP":
            break
        try:
            chunk_filename = task["filename"]
            chunk_path = task["chunk_path"]
            txt_output_path = task["txt_path"]
            task_id = task["task_id"]

            logging.info(f"DistillWorker-{worker_id}: Processing distill task {chunk_filename}")

            with open(txt_output_path, "r", encoding="utf-8-sig") as f:
                transcription = f.read()

            run_distill_process(task_id, chunk_filename, chunk_path, transcription)
            logging.info(f"DistillWorker-{worker_id}: Completed distill task {chunk_filename}")
        except Exception as e:
            logging.error(f"DistillWorker-{worker_id}: Distill task failed: {e}")
        finally:
            distill_task_queue.task_done()


def apply_glossary_fix(text: str, glossary: dict) -> str:
    """
    套用 glossary 的 common_errors 修正，修補常見錯字。
    """
    fixed_text = text
    for wrong, right in glossary.get("common_errors", {}).items():
        fixed_text = re.sub(wrong, right, fixed_text)
    return fixed_text


def run_distill_process(task_id: str, chunk_filename: str, chunk_path: str, transcription: str):
    """
    用 WhisperPool 再跑一遍 chunk，本地轉錄，與 Gemini 結果做差異比對，輸出 distill.json。
    """
    local_whisper_text = None
    try:
        logging.info(f"Parallel STT Distill: WhisperPool {chunk_filename}")
        local_whisper_text = WhisperPool.transcribe(chunk_path)
        local_whisper_text = apply_glossary_fix(local_whisper_text, {"common_errors": {}})

        logging.info(f"Parallel STT Distill: {chunk_filename}")

        if local_whisper_text and transcription:
            from difflib import SequenceMatcher

            s = SequenceMatcher(None, local_whisper_text, transcription)
            diff_items = []
            for tag, i1, i2, j1, j2 in s.getopcodes():
                if tag != "replace":
                    continue
                lw = local_whisper_text[i1:i2]
                gr = transcription[j1:j2]
                if 0 < len(lw) <= 100 and 0 < len(gr) <= 100:
                    diff_items.append({"local_whisper_raw": lw, "gemini_corrected": gr})

            distill_dir = Path("A:/dataset/distill")
            distill_dir.mkdir(parents=True, exist_ok=True)
            chunk_distill_path = distill_dir / f"{task_id}_{chunk_filename.replace('.wav', '')}_distill.json"
            atomic_json_dump(
                {
                    "task_id": task_id,
                    "chunk_filename": chunk_filename,
                    "diff_mappings": diff_items,
                    "local_whisper_text": local_whisper_text,
                    "gemini_text": transcription,
                    "processed_at": time.strftime("%Y-%m-%d %H:%M:%S"),
                },
                str(chunk_distill_path),
            )
            logging.info(f"Parallel STT Distill: saved {chunk_distill_path}")
    except Exception as de:
        logging.warning(f"Parallel STT Distill: {de}")


# ── 單一 chunk 處理邏輯：Gemini / Vertex / 回退 / retry ──


def process_single_chunk(
    chunk: dict,
    client,
    qm: QuotaManager | None,
    current_key_ref: list,
    key_lock: threading.Lock,
    task_id: str,
    chunks_manifest_path: str,
    config: dict,
    chunk_errors_log: str,
    stt_engine: str,
    manifest_update_queue: queue.Queue,
    distill_task_queue: queue.Queue,
) -> bool:
    """
    真正的 STT chunk 處理主程式。
    - 會根據 stt_engine 選擇 Gemini 或 Vertex AI。
    - 會對 429 / ResourceExhausted / 503 等做 backoff與條件 retry。
    - 成功時把 status=completed，失敗達上限後標 failed。
    - 失敗後 run_stt 會把整體 manifest 打回 chunked，並重設 chunk 為 pending。
    """
    chunk_path = chunk["chunk_path"]
    chunk_filename = chunk["filename"]
    retry_limit = 6
    uploaded_file = None
    upload_client = None
    last_upload_key = None
    consecutive_429_count = 0
    processing_sent = False

    logging.info(f"Parallel STT: Processing chunk {chunk_filename}")

    def queue_processing_once():
        nonlocal processing_sent
        if not processing_sent:
            now_ts = time.time()
            manifest_update_queue.put(
                {
                    "filename": chunk_filename,
                    "status": "processing",
                    "started_at": now_ts,
                    "updated_at": now_ts,
                    "retry_count": chunk.get("retry_count", 0),
                    "last_error": None,
                }
            )
            processing_sent = True

    def queue_retrying(error_obj: str):
        now_ts = time.time()
        manifest_update_queue.put(
            {
                "filename": chunk_filename,
                "status": "retrying",
                "updated_at": now_ts,
                "retry_count": chunk.get("retry_count", 0),
                "last_error": str(error_obj),
            }
        )

    def queue_failed(error_obj: str):
        now_ts = time.time()
        manifest_update_queue.put(
            {
                "filename": chunk_filename,
                "status": "failed",
                "updated_at": now_ts,
                "failed_at": now_ts,
                "retry_count": chunk.get("retry_count", 0),
                "last_error": str(error_obj),
            }
        )

    def queue_completed():
        now_ts = time.time()
        manifest_update_queue.put(
            {
                "filename": chunk_filename,
                "status": "completed",
                "updated_at": now_ts,
                "retry_count": chunk.get("retry_count", 0),
                "last_error": None,
            }
        )

    chunk_status = chunk.get("status", "pending")
    chunk_retry_count = chunk.get("retry_count", 0)

    while chunk_retry_count < retry_limit:
        try:
            queue_processing_once()

            # Gemini / Vertex 上傳與轉錄
            if stt_engine in ("gemini", "vertexai"):
                filesize = os.path.getsize(chunk_path)
                if filesize > 2_000 * 1024 * 1024:
                    raise ValueError(f"{chunk_filename} 超過 2GB 上限")

                if stt_engine == "gemini":
                    with key_lock:
                        active_key = current_key_ref[0]
                    if upload_client is not None and last_upload_key != active_key:
                        uploaded_file = None
                        upload_client = None
                        last_upload_key = None
                    if not uploaded_file:
                        with key_lock:
                            active_key = current_key_ref[0]
                        upload_client = genai.Client(api_key=active_key)
                        last_upload_key = active_key
                        with open(chunk_path, "rb") as f:
                            audiodata = f.read()
                        uploaded_file = types.Part.from_bytes(
                            data=audiodata,
                            mimetype="audio/wav",
                        )
                    transcription = call_with_timeout(
                        transcribe_chunk,
                        API_TRANSCRIBE_TIMEOUT,
                        upload_client,
                        uploaded_file,
                        config,
                        qm,
                        active_key,
                    )
                else:
                    # Vertex AI 模式
                    vertex_project = config.get("api", {}).get("vertexai_project")
                    with open(chunk_path, "rb") as f:
                        audiodata = f.read()
                    uploaded_file = types.Part.from_bytes(
                        data=audiodata,
                        mimetype="audio/wav",
                    )
                    upload_client = genai.Client(
                        vertexai=True,
                        project=vertex_project,
                        location="us-central1",
                    )
                    transcription = call_with_timeout(
                        transcribe_chunk,
                        API_TRANSCRIBE_TIMEOUT,
                        upload_client,
                        uploaded_file,
                        config,
                        None,
                        None,
                    )

                # 存檔
                txt_output_path = os.path.splitext(chunk_path)[0] + ".txt"
                with open(txt_output_path, "w", encoding="utf-8-sig") as tf:
                    tf.write(transcription or "")

                json_output_path = os.path.splitext(chunk_path)[0] + ".json"
                atomic_json_dump(
                    {
                        "task_id": task_id,
                        "filename": chunk_filename,
                        "part_no": chunk.get("part_no"),
                        "chunk_id": chunk.get("chunk_id"),
                        "transcription": transcription,
                        "processed_at": time.strftime("%Y-%m-%d %H:%M:%S"),
                    },
                    json_output_path,
                )

                distill_task_queue.put(
                    {
                        "filename": chunk_filename,
                        "chunk_path": chunk_path,
                        "txt_path": txt_output_path,
                        "task_id": task_id,
                    }
                )

                chunk_status = "completed"
                consecutive_429_count = 0
                if stt_engine == "gemini" and qm is not None:
                    qm.reset_key(current_key_ref[0])
                logging.info(f"Parallel STT OK: {chunk_filename}")
                queue_completed()
                return True

        except Exception as e:
            err_str = str(e)
            is_429 = (
                hasattr(e, "name")
                and "ResourceExhausted" in getattr(e, "name", "")
            ) or "429" in err_str

            if stt_engine == "gemini" and is_429:
                consecutive_429_count += 1
                chunk_retry_count += 1
                queue_retrying(e)
                with key_lock:
                    current_key = current_key_ref[0]
                # 讓 QuotaManager 知道 429 錯誤，切換 key
                try:
                    res = qm.handle_error(e, current_key, consecutive_429_count)
                    new_key = res.get("new_key")
                except Exception:
                    new_key = None
                if new_key:
                    with key_lock:
                        current_key_ref[0] = new_key
                    consecutive_429_count = 0
                    uploaded_file = None
                else:
                    queue_failed(e)
                    break
                # 休息一下再重試
                time.sleep(res.get("sleep_time", 5))
                continue

            if stt_engine == "vertexai" and hasattr(e, "name") and "ResourceExhausted" in getattr(e, "name", ""):
                chunk_retry_count += 1
                queue_retrying(e)
                time.sleep(15)
                continue

            if any(
                key in err_str
                for key in ("503", "UNAVAILABLE", "Server disconnected")
            ):
                backoff_times = [5, 10, 15, 30, 30, 30]
                wait_503 = backoff_times[min(chunk_retry_count, len(backoff_times) - 1)]
                chunk_retry_count += 1
                with open(chunk_errors_log, "a", encoding="utf-8") as ef:
                    ef.write(
                        f"{time.strftime('%Y-%m-%d %H:%M:%S')} {task_id} {chunk_filename} 503/UNAVAILABLE {e}\n"
                    )
                if chunk_retry_count >= retry_limit:
                    queue_failed(e)
                else:
                    queue_retrying(e)
                    time.sleep(wait_503)
                continue

            # 其他錯誤
            import traceback

            tb_str = traceback.format_exc().replace("\n", "\\n")
            with open(chunk_errors_log, "a", encoding="utf-8") as ef:
                ef.write(
                    f"{time.strftime('%Y-%m-%d %H:%M:%S')} "
                    f"{task_id} {chunk_filename} Other Error "
                    f"ExceptionClass={getattr(e, 'name', type(e).__name__)} "
                    f"RetryCount={chunk_retry_count} Error={e} Traceback={tb_str}\n"
                )

            chunk_retry_count += 1
            if chunk_retry_count >= retry_limit:
                queue_failed(f"chunk failed after retries: {chunk_filename}")
            else:
                queue_retrying(e)

            return False

    return True

def transcribe_chunk(client, uploaded_file, config, qm=None, api_key=None) -> str:
    """
    呼叫 Gemini 或 Vertex 的文字轉錄 API。
    保留你原來的 prompt、safety_settings 與 router 寫入行為。
    """
    router = getrouter()
    model = router.acquire()

    prompt = (
        "你是一位專業的台灣法律課程與講座的逐字稿整理助理。\n"
        "請將提供的音檔進行精確的繁體中文語音轉錄，並遵守以下規定：\n"
        "【重要：完整性與精確性要求】\n"
        "此檔案為極為關鍵的法律教學教材，是後續所有 AI 專案與智能庫的智慧基礎。"
        "請務必做到「一字不漏、完全轉錄」！\n"
        "嚴禁任何摘要、縮寫、省略或簡化發言的行為。\n"
        "1. 輸出格式一律為繁體中文（台灣習慣用詞）。\n"
        "2. 保留台灣法律專業術語。\n"
        "3. 完整保留所有提到的法律法條名稱與條號。\n"
        "4. 保留釋字字號與法院判決字號之標準格式。\n"
        "5. 聽不清楚或不確定之處不可憑空預測，一律以 [待確認] 標記。\n"
        "6. 必須定期插入時間戳記（例如 [01:23]、[05:45]）。\n"
        "7. 辨識到章節/主題切換時插入 Markdown 標題。\n"
        "8. 辨識到「爭點」、「重點整理」時加粗標記。\n"
        "9. 適度過濾贅字，使文字流暢。\n"
    )

    if qm is not None and api_key is not None:
        try:
            qm.throttle_key(api_key)
        except Exception:
            pass

    try:
        from google.genai import types as genai_types

        safety_settings = [
            genai_types.SafetySetting(
                category=genai_types.HarmCategory.HARM_CATEGORY_HATE_SPEECH,
                threshold=genai_types.HarmBlockThreshold.BLOCK_ONLY_HIGH,
            ),
            genai_types.SafetySetting(
                category=genai_types.HarmCategory.HARM_CATEGORY_HARASSMENT,
                threshold=genai_types.HarmBlockThreshold.BLOCK_ONLY_HIGH,
            ),
            # 以下兩個類別在 prompt_for_coder.txt 中已被 REDACTED，這裡不展開具體名稱
        ]

        response = client.models.generate_content(
            model=model,
            contents=[uploaded_file, prompt],
            config=genai_types.GenerateContentConfig(
                safety_settings=safety_settings,
                temperature=0.3,
            ),
        )
        router.report_success(model)
        text = response.text
        if not text or not text.strip():
            candidates = response.candidates or []
            finish = str(
                getattr(candidates[0], "finish_reason", "UNKNOWN")
            ) if candidates else "NO_CANDIDATES"
            if "SAFETY" in finish:
                raise ValueError(
                    f"Safety filter triggered (finish_reason={finish})，換模型重試"
                )
            raise ValueError(
                f"API 回傳空白內容 (finish_reason={finish})，觸發 retry"
            )
        return text
    except Exception as e:
        router.report_error(model, e)
        raise


# ── 主函式：run_stt(task_id, exclusive_key=None) ──


def run_stt(task_id: str, exclusive_key: str | None = None) -> bool:
    """
    入口：讓 run_workflow 呼叫，對整個 task 的所有 chunk 做 STT。

    行為：
    - 讀取 config.yaml 與 paths（透過 workflow_helper）。
    - 根據 settings.stt_engine 選擇 gemini 或 vertexai。
    - 啟動背景 manifest writer 與 distill workers。
    - 使用 ThreadPoolExecutor 以 CHUNK_CONCURRENCY 併發執行 process_single_chunk。
    - 最後根據 all_succeeded 將整體 manifest 狀態更新為：
      - 成功：status=transcribed, steps["stt"]="completed"
      - 失敗：status=chunked, steps["stt"]="pending"，並重置失敗 chunk。
    """
    from workflow_helper import load_config, get_resolved_paths

    config = load_config()
    paths = get_resolved_paths(config)

    manifests_dir = paths["manifests_dir"]
    workflow_log = os.path.join(paths["logs_dir"], "workflow.log")
    chunk_errors_log = os.path.join(paths["logs_dir"], "chunk_errors.log")

    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s [%(levelname)s] %(message)s",
        handlers=[
            logging.StreamHandler(sys.stdout),
            logging.FileHandler(workflow_log, encoding="utf-8"),
        ],
    )
    os.makedirs(os.path.dirname(chunk_errors_log), exist_ok=True)

    manifest_path = os.path.join(manifests_dir, f"{task_id}.json")
    if not os.path.exists(manifest_path):
        return False

    with open(manifest_path, "r", encoding="utf-8-sig") as f:
        manifest = json.load(f)

    chunks_manifest_path = os.path.join(manifests_dir, f"{task_id}_chunks.json")
    if not os.path.exists(chunks_manifest_path):
        return False

    with open(chunks_manifest_path, "r", encoding="utf-8-sig") as f:
        chunks = json.load(f)

    if not chunks:
        return False

    stt_engine = (config.get("settings", {}) or {}).get("stt_engine", "gemini")

    qm = None
    current_key = None
    client = None
    key_lock = threading.Lock()

    if stt_engine == "gemini":
        # 保留 QuotaManager 排他 key 語義
        try:
            qm = QuotaManager()
        except Exception as qm_err:
            logging.error(f"QuotaManager init failed: {qm_err}")
            qm = None

        if exclusive_key:
            current_key = exclusive_key
        elif qm is not None:
            current_key = qm.acquire_key_exclusive()
        else:
            current_key = None

        if current_key:
            client = genai.Client(api_key=current_key)

    elif stt_engine == "vertexai":
        # Vertex 模式：不使用 QuotaManager，也不申請 Gemini key
        vertex_project = (config.get("api", {}) or {}).get("vertexai_project")
        vertex_location = (config.get("settings", {}) or {}).get(
            "vertexai_location", "us-central1"
        )
        client = genai.Client(vertexai=True, project=vertex_project, location=vertex_location)
        qm = None
        current_key = None

    current_key_ref = [current_key]
    pending_chunks = [c for c in chunks if c.get("status") != "completed"]

    manifest_update_queue: queue.Queue = queue.Queue()
    distill_task_queue: queue.Queue = queue.Queue()

    # 啟動背景 writer（Bug 1 修補：避免 I/O 阻塞）
    writer_thread = threading.Thread(
        target=manifest_writer_thread,
        args=(chunks_manifest_path, manifest_update_queue),
        daemon=True,
    )
    writer_thread.start()

    # 啟動背景 distill workers（Bug 2 修補：限制最大併發）
    distill_workers = []
    for i in range(6):
        t = threading.Thread(
            target=distill_worker_thread,
            args=(i, distill_task_queue),
            daemon=True,
        )
        t.start()
        distill_workers.append(t)

    all_succeeded = True

    with ThreadPoolExecutor(max_workers=CHUNK_CONCURRENCY) as executor:
        futures = {
            executor.submit(
                process_single_chunk,
                chunk,
                client,
                qm,
                current_key_ref,
                key_lock,
                task_id,
                chunks_manifest_path,
                config,
                chunk_errors_log,
                stt_engine,
                manifest_update_queue,
                distill_task_queue,
            ): chunk
            for chunk in pending_chunks
        }
        for future in as_completed(futures):
            chunk = futures[future]
            try:
                if not future.result():
                    all_succeeded = False
            except Exception as exc:
                chunk["status"] = "failed"
                all_succeeded = False

    # 等待 writer 把所有更新寫完，再發 STOP，避免最後批次遺失
    manifest_update_queue.join()
    manifest_update_queue.put("STOP")

    for _ in range(6):
        distill_task_queue.put("STOP")

    writer_thread.join()
    for t in distill_workers:
        t.join()

    # ── 更新整體 manifest 與 chunks ──
    if all_succeeded:
        manifest["status"] = "transcribed"
        manifest.setdefault("steps", {})
        manifest["steps"]["stt"] = "completed"
        atomic_json_dump(manifest, manifest_path)
        atomic_json_dump(chunks, chunks_manifest_path)
        logging.info(f"STT for task {task_id} completed; manifest marked transcribed.")
    else:

def _build_dispatch_batch(active_tasks: list[tuple[Path, dict]], chunking_only: bool) -> tuple[list[tuple[Path, dict]], int, str]:

    queued_tasks = [(p, d) for p, d in active_tasks if d.get("status") == "queued"]
    chunked_tasks = [(p, d) for p, d in active_tasks if d.get("status") == "chunked"]

    if chunking_only:
        batch_type = "queued"
        selected = queued_tasks
        concurrency = PREPROCESS_CONCURRENCY

        batch_type = "chunked"

        batch_type = "queued"
        concurrency = PREPROCESS_CONCURRENCY
        return queued_tasks[:concurrency * 8], concurrency, batch_type

    return [], PREPROCESS_CONCURRENCY, "none"
