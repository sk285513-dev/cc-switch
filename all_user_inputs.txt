                    "Future [%s/%s] raised exception: %s",
                    tid, cid, exc,
                )

import contextlib
import logging
import re
import sqlite3
from pathlib import Path
from typing import Any

DB_PATH = Path("tasks.db")
logger = logging.getLogger(__name__)


def _get_conn(timeout: float = 30.0) -> sqlite3.Connection:
    """
    Factory: WAL mode + foreign keys.
    Always use via `with contextlib.closing(_get_conn()) as conn:`.
    """
    conn = sqlite3.connect(str(DB_PATH), timeout=timeout)
    conn.execute("PRAGMA journal_mode=WAL;")
    conn.execute("PRAGMA synchronous=NORMAL;")
    conn.execute("PRAGMA foreign_keys=ON;")
    conn.row_factory = sqlite3.Row
    return conn


def init_db() -> None:
    """建立資料表（冪等）。"""
    with contextlib.closing(_get_conn()) as conn:
        conn.executescript("""
            CREATE TABLE IF NOT EXISTS tasks (
                task_id     TEXT PRIMARY KEY,
                file_name   TEXT NOT NULL,
                status      TEXT NOT NULL DEFAULT 'pending',
                total_chunks INTEGER NOT NULL DEFAULT 0,
                created_at  TEXT NOT NULL DEFAULT (datetime('now')),
                updated_at  TEXT NOT NULL DEFAULT (datetime('now'))
            );

            CREATE TABLE IF NOT EXISTS chunks (
                chunk_id    TEXT NOT NULL,
                task_id     TEXT NOT NULL,
                status      TEXT NOT NULL DEFAULT 'pending',
                retry_count INTEGER NOT NULL DEFAULT 0,
                last_error  TEXT,
                updated_at  TEXT NOT NULL DEFAULT (datetime('now')),
                PRIMARY KEY (chunk_id, task_id),
                FOREIGN KEY (task_id) REFERENCES tasks(task_id) ON DELETE CASCADE
            );

            CREATE INDEX IF NOT EXISTS idx_chunks_task_status
                ON chunks(task_id, status);
        """)
        conn.commit()

# DEFECT FIX #1 — UPSERT 漏掉 task_id 導致提早合併

def update_chunk_status(
    task_id: str,
    chunk_id: str,
    status: str,
    error: str | None = None,
) -> None:
    """
    UPSERT chunk 狀態。
    PRIMARY KEY = (chunk_id, task_id)，兩欄缺一不可。
    """
    with contextlib.closing(_get_conn()) as conn:
        with conn:          # BEGIN / COMMIT / ROLLBACK on exception
            conn.execute(
                """
                INSERT INTO chunks (chunk_id, task_id, status, last_error, updated_at)
                VALUES (:chunk_id, :task_id, :status, :error, datetime('now'))
                ON CONFLICT(chunk_id, task_id) DO UPDATE SET
                    status     = excluded.status,
                    last_error = excluded.last_error,
                    updated_at = excluded.updated_at
                """,
                {
                    "chunk_id": chunk_id,
                    "task_id":  task_id,   # ← 修復：不再遺漏
                    "status":   status,
                    "error":    error,
                },
            )
            _sync_task_completion(conn, task_id)


def _sync_task_completion(conn: sqlite3.Connection, task_id: str) -> None:
    """同一 transaction 內：若所有 chunk done → 更新 task.status。"""
    row = conn.execute(
        """
        SELECT
            COUNT(*)                                    AS total,
            SUM(CASE WHEN status = 'done' THEN 1 ELSE 0 END) AS done_cnt
        FROM chunks WHERE task_id = ?
        """,
        (task_id,),
    ).fetchone()

    if row and row["total"] > 0 and row["total"] == row["done_cnt"]:
        conn.execute(
            "UPDATE tasks SET status='completed', updated_at=datetime('now') WHERE task_id=?",
            (task_id,),
        )

# DEFECT FIX #3 — reset_stuck_chunk 必須遞增 retry_count
# ──────────────────────────────────────────────
def reset_stuck_chunk(
    task_id: str,
    chunk_id: str,
    max_retries: int = 5,
) -> bool:
    """
    將 `processing` 狀態的 chunk 重置為 `pending`。
    - 每次呼叫 retry_count += 1
    - 超過 max_retries 則標記為 failed，防止無限失敗迴圈
    回傳 True = 成功重置；False = 已達上限，標記 failed
    """
    with contextlib.closing(_get_conn()) as conn:
        with conn:
            row = conn.execute(
                "SELECT retry_count FROM chunks WHERE chunk_id=? AND task_id=?",
                (chunk_id, task_id),
            ).fetchone()

            if row is None:
                logger.warning("reset_stuck_chunk: chunk %s/%s not found", task_id, chunk_id)
                return False

            new_count = row["retry_count"] + 1

            if new_count > max_retries:
                conn.execute(
                    """
                    UPDATE chunks
                    SET status='failed',
                        retry_count=:cnt,
                        last_error='Exceeded max retries',
                        updated_at=datetime('now')
                    WHERE chunk_id=:cid AND task_id=:tid
                    """,
                    {"cnt": new_count, "cid": chunk_id, "tid": task_id},
                )
                logger.error(
                    "chunk %s/%s exceeded max_retries=%d → marked failed",
                    task_id, chunk_id, max_retries,
                )
                return False

            conn.execute(
                """
                UPDATE chunks
                SET status='pending',
                    retry_count=:cnt,
                    updated_at=datetime('now')
                WHERE chunk_id=:cid AND task_id=:tid AND status='processing'
                """,
                {"cnt": new_count, "cid": chunk_id, "tid": task_id},
            )
            return True

def get_active_tasks() -> list[dict[str, Any]]:
